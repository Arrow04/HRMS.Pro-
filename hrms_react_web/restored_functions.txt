  // Fetch employees for department head dropdown - only when needed

    queryKey: ['employees'],
    queryFn: async () => {
      const response = await api.get('/employees');

      return data.filter((item: any) => item.status !== 'inactive' && item.is_active !== false);

  });

      const matchCompany = companyFilter === 'all' || item.companyId?.toString() === companyFilter.toString();
      const matchBranch = branchFilter === 'all' || item.branchId?.toString() === branchFilter.toString();
      const matchDepartment = departmentFilter === 'all' || item.departmentId?.toString() === departmentFilter.toString();
      const matchStatus = statusFilter === 'all' || item.status === statusFilter;
      return matchSearch && matchCompany && matchBranch && matchDepartment && matchStatus;
    });

      setFormData(INITIAL_FORM_DATA);
      setFormTab('basic');

        setIsSearchingEmployees(false);

  const gradeValues = Array.isArray(designationGradeOptions) ? designationGradeOptions : [];

      case 'branches': return Array.isArray(branchStatusOptions) ? branchStatusOptions : [];
      case 'departments': return Array.isArray(departmentStatusOptions) ? departmentStatusOptions : [];
      case 'designations': return Array.isArray(designationStatusOptions) ? designationStatusOptions : [];
      default: return [];

  const statusArray = getStatusArray();

  const createMutation = useMutation({
    mutationFn: async (data: any) => {
      return api.post(`/${activeTab}`, data);
    },
    onSuccess: () => {
      toast.success(`${getSingularName(activeTab)} created successfully`);
      queryClient.invalidateQueries({ queryKey: [activeTab] });
      queryClient.invalidateQueries({ queryKey: ['companyStats'] });
      setShowModal(false);
      setFormData(INITIAL_FORM_DATA);
    },
    onError: (error: any) => {
      toast.error(`Failed to create ${getSingularName(activeTab)}: ${error.response?.data?.message || error.message}`);
    }
  });

  // Update mutation
  const updateMutation = useMutation({
    mutationFn: async ({ id, data }: { id: number; data: any }) => {
      return api.put(`/${activeTab}/${id}`, data);
    },
    onSuccess: () => {
      toast.success(`${getSingularName(activeTab)} updated successfully`);
      queryClient.invalidateQueries({ queryKey: [activeTab] });
      setShowModal(false);
      setEditingItem(null);
      setFormData(INITIAL_FORM_DATA);
      setSearchTerm('');
      setCompanyFilter('all');
    },
    onError: () => {
      toast.error(`Failed to update ${getSingularName(activeTab)}`);
    }
  });

  // Delete mutation
  const deleteMutation = useMutation({
    mutationFn: async (id: number) => {
      return api.delete(`/${activeTab}/${id}`);
    },
    onSuccess: () => {
      toast.success(`${getSingularName(activeTab)} deleted successfully`);
      queryClient.invalidateQueries({ queryKey: [activeTab] });
      queryClient.invalidateQueries({ queryKey: ['companyStats'] });
    },
    onError: () => {
      toast.error(`Failed to delete ${getSingularName(activeTab)}`);
    }
  });

  // Bulk upload mutation
  const bulkUploadMutation = useMutation({
    mutationFn: async (file: File) => {
      const formData = new FormData();
      formData.append('file', file);
      return api.post(`/${activeTab}/bulk-upload`, formData, {
        headers: { 'Content-Type': 'multipart/form-data' }
      });
    },
    onSuccess: (data: any) => {
      const created = data.data?.created || 0;
      const updated = data.data?.updated || 0;
      toast.success(`Bulk upload completed: ${created} created, ${updated} updated`);
      queryClient.invalidateQueries({ queryKey: [activeTab] });
      queryClient.invalidateQueries({ queryKey: ['companyStats'] });

      setUploadFile(null);

      toast.error(error.response?.data?.message || 'Bulk upload failed');

  const downloadTemplate = async () => {
    try {
      const response = await api.get(`/${activeTab}/template`, { responseType: 'blob' });
      const url = window.URL.createObjectURL(new Blob([response.data]));
      const link = document.createElement('a');
      link.href = url;
      link.setAttribute('download', `${getSingularName(activeTab)}_bulk_upload_template.csv`);
      document.body.appendChild(link);
      link.click();
      link.remove();
      window.URL.revokeObjectURL(url);
      toast.success('Template downloaded successfully');
    } catch (error) {
      toast.error('Failed to download template');
    }
  };

    try {
      const response = await api.get(`/${activeTab}/export`, { responseType: 'blob' });
      const url = window.URL.createObjectURL(new Blob([response.data]));
      const link = document.createElement('a');
      link.href = url;
      link.setAttribute('download', `${activeTab}_export.csv`);
      document.body.appendChild(link);
      link.click();
      link.remove();
      window.URL.revokeObjectURL(url);
      toast.success('Data exported successfully');
    } catch (error) {
      toast.error('Failed to export data');
    }
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    let dataToSubmit: any = { code: formData.code, description: formData.description, status: formData.status };
    if (activeTab !== 'designations') dataToSubmit.name = formData.name;
    
    if (activeTab === 'companies') {
      dataToSubmit = { ...dataToSubmit, registrationNumber: formData.registrationNumber, taxId: formData.taxId, industry: formData.industry, companySize: formData.companySize, address: formData.address };
    } else if (activeTab === 'branches') {
      dataToSubmit = { ...dataToSubmit, location: formData.location, latitude: formData.latitude, longitude: formData.longitude, geofenceRadius: formData.geofenceRadius, companyId: formData.companyId };
    } else if (activeTab === 'departments') {
      dataToSubmit = { ...dataToSubmit, companyId: formData.companyId, managerId: formData.managerId };
    } else if (activeTab === 'designations') {
      dataToSubmit = { ...dataToSubmit, companyId: formData.companyId, title: formData.title, grade: formData.grade };
    }
    
    if (editingItem) updateMutation.mutate({ id: editingItem.id, data: dataToSubmit });
    else createMutation.mutate(dataToSubmit);
  };

  const handleToggleStatus = async (item: any) => {
    try {
      const newStatus = item.status === 'active' ? 'inactive' : 'active';
      await updateMutation.mutateAsync({ id: item.id, data: { ...item, status: newStatus } });
      toast.success(`Status updated successfully`);
    } catch (error) {
      toast.error('Failed to update status');
    }
  };

  return (
    <div className="min-h-screen bg-[#F8FAFC] p-6">
      <div className="max-w-[1920px] mx-auto space-y-6">
        {/* Header */}
        <div className="flex flex-col lg:flex-row lg:items-center lg:justify-between gap-4 mb-8">
          <div>
            <h1 className="text-2xl font-bold text-[#0F172A]">Company Structure</h1>
            <p className="text-sm text-[#64748B] mt-1">Manage companies, branches, departments and designations</p>
          </div>
          <div className="flex items-center gap-3">
            <button onClick={exportData} className="flex items-center gap-2 px-4 py-2.5 bg-white border border-[#E2E8F0] text-[#0F172A] text-sm font-medium rounded-xl hover:bg-[#F8FAFC] transition-colors">
              <Download className="w-4 h-4" />
              <span className="hidden sm:inline">Export</span>
            </button>
            <button onClick={() => setShowBulkUpload(true)} className="flex items-center gap-2 px-4 py-2.5 bg-white border border-[#E2E8F0] text-[#0F172A] text-sm font-medium rounded-xl hover:bg-[#F8FAFC] transition-colors">
              <Upload className="w-4 h-4" />
              <span className="hidden sm:inline">Bulk Upload</span>
            </button>
            <button onClick={() => {
              setEditingItem(null);
              setFormData(INITIAL_FORM_DATA);
              setShowModal(true);
            }} className="flex items-center gap-2 px-4 py-2.5 bg-[#1C64F2] text-white text-sm font-medium rounded-xl hover:bg-[#1E40AF] transition-all duration-200 shadow-sm shadow-[#1C64F2]/20">
              <Plus className="w-4 h-4" />
              <span className="hidden sm:inline">Add New</span>
            </button>
          </div>
        </div>

        {/* TABS - Pill Style */}
        <div className="bg-white rounded-2xl border border-[#E2E8F0] p-4 mb-6">
          <div className="flex flex-wrap items-center gap-2">
            {TABS.map((tab) => (
              <button
                key={tab.id}
                onClick={() => setActiveTab(tab.id as any)}
                className={`flex items-center gap-2 px-4 py-2.5 rounded-lg font-medium text-sm transition-all duration-200 whitespace-nowrap ${
                  activeTab === tab.id ? 'bg-[#1C64F2] text-white shadow-md shadow-[#1C64F2]/20' : 'text-[#64748B] hover:text-[#0F172A] hover:bg-[#F8FAFC]'
                }`}
              >
                <tab.icon className="w-4 h-4" />
                {tab.label}
              </button>
            ))}
          </div>
        </div>

        <div className="bg-white rounded-2xl border border-[#E2E8F0] p-6">
          {/* Filter Row */}
          <div className="flex flex-col gap-4 mb-6">
            <div className="flex flex-col sm:flex-row items-start sm:items-center gap-4">
              <div className="relative flex-1 w-full">
                <Search className="absolute left-4 top-1/2 -translate-y-1/2 w-5 h-5 text-[#64748B]" />
                <input
                  type="text"
                  placeholder="Search..."
                  value={searchTerm}
                  onChange={(e) => setSearchTerm(e.target.value)}
                  className="w-full pl-11 pr-4 py-2.5 bg-white border border-[#E2E8F0] rounded-xl text-sm focus:outline-none focus:ring-2 focus:ring-[#1C64F2] transition-shadow"
                />
              </div>
              <div className="flex items-center gap-3 w-full sm:w-auto">
                <DateRangePicker
                  startDate={startDate}
                  endDate={endDate}
                  onDateChange={(start, end) => {
                    setStartDate(start);
                    setEndDate(end);

                <button
                  onClick={() => {
                    setSearchTerm('');
                    setCompanyFilter('all');
                    setBranchFilter('all');
                    setDepartmentFilter('all');
                    setStatusFilter('all');
                    setStartDate('');
                    setEndDate('');
                  }}
                  className={`flex items-center gap-2 px-4 py-2.5 rounded-xl text-sm font-medium transition-colors whitespace-nowrap ${ searchTerm !== '' || companyFilter !== 'all' || branchFilter !== 'all' || departmentFilter !== 'all' || statusFilter !== 'all' || startDate !== '' ? 'bg-[#C81E1E] text-white hover:bg-[#A01515]' : 'bg-[#64748B] text-white hover:bg-[#475569]' }`}
                >
                  <X className="w-4 h-4" />
